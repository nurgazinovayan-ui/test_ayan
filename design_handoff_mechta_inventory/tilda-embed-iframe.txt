<script>
(function(){
  var f = document.getElementById('mechta-inv-frame');
  function fit(){
    try {
      var d = f.contentDocument; if (!d || !d.body) return;
      var h = Math.max(d.body.scrollHeight, d.documentElement.scrollHeight);
      if (h > 200) f.style.height = (h + 24) + 'px';
    } catch(e){}
  }
  f.addEventListener('load', function(){ fit(); setTimeout(fit, 400); });
  setInterval(fit, 700);
})();
